Инструкция по настройке

1. Зарегистрируйтесь на supabase.com

2. Создайте новый проект и запомните пароль

3. В настройках API скопируйте Project URL и anon public key

4. Вставьте их в файлы BroSky (index.html, catalog.html, и т.д.)

5. Выполните SQL код ниже для создания таблиц

SQL код для создания таблиц

-- ============================================
-- Таблица объявлений (listings)
-- ============================================
CREATE TABLE IF NOT EXISTS listings (
    id SERIAL PRIMARY KEY,
    title TEXT NOT NULL,
    city TEXT NOT NULL,
    price INT NOT NULL,
    beds INT DEFAULT 1,
    baths INT DEFAULT 1,
    guests INT DEFAULT 4,
    type TEXT,
    description TEXT,
    image TEXT,
    rating DECIMAL(3,1) DEFAULT 4.5,
    reviews INT DEFAULT 42,
    created_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- Таблица пользователей (users)
-- ============================================
CREATE TABLE IF NOT EXISTS users (
    id SERIAL PRIMARY KEY,
    name TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    password TEXT NOT NULL,
    phone TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- Таблица бронирований (bookings)
-- ============================================
CREATE TABLE IF NOT EXISTS bookings (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    listing_id INT REFERENCES listings(id) ON DELETE CASCADE,
    checkin DATE NOT NULL,
    checkout DATE NOT NULL,
    guests INT DEFAULT 2,
    days INT,
    total_price INT,
    status TEXT DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- Таблица заявок (pending_listings)
-- ============================================
CREATE TABLE IF NOT EXISTS pending_listings (
    id BIGINT PRIMARY KEY,
    title TEXT NOT NULL,
    city TEXT NOT NULL,
    type TEXT,
    address TEXT,
    price INT NOT NULL,
    beds INT,
    baths INT,
    guests INT,
    description TEXT,
    amenities TEXT,
    image TEXT,
    status TEXT DEFAULT 'pending',
    created_at TIMESTAMP,
    userId INT,
    userName TEXT,
    userEmail TEXT
);
