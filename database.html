-- ============================================
-- 1. Таблица объявлений (listings)
-- ============================================
CREATE TABLE IF NOT EXISTS listings (
    id SERIAL PRIMARY KEY,
    title TEXT NOT NULL,
    city TEXT NOT NULL,
    price INT NOT NULL,
    beds INT DEFAULT 1,
    baths INT DEFAULT 1,
    guests INT DEFAULT 4,
    type TEXT,
    description TEXT,
    image TEXT,
    rating DECIMAL(3,1) DEFAULT 4.5,
    reviews INT DEFAULT 42,
    created_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- 2. Таблица пользователей (users)
-- ============================================
CREATE TABLE IF NOT EXISTS users (
    id SERIAL PRIMARY KEY,
    name TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    password TEXT NOT NULL,
    phone TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- 3. Таблица бронирований (bookings)
-- ============================================
CREATE TABLE IF NOT EXISTS bookings (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    listing_id INT REFERENCES listings(id) ON DELETE CASCADE,
    checkin DATE NOT NULL,
    checkout DATE NOT NULL,
    guests INT DEFAULT 2,
    days INT,
    total_price INT,
    status TEXT DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT NOW()
);

-- ============================================
-- 4. Таблица заявок на добавление (pending_listings)
-- ============================================
CREATE TABLE IF NOT EXISTS pending_listings (
    id BIGINT PRIMARY KEY,
    title TEXT NOT NULL,
    city TEXT NOT NULL,
    type TEXT,
    address TEXT,
    price INT NOT NULL,
    beds INT,
    baths INT,
    guests INT,
    description TEXT,
    amenities TEXT,
    image TEXT,
    status TEXT DEFAULT 'pending',
    created_at TIMESTAMP,
    userId INT,
    userName TEXT,
    userEmail TEXT
);

-- ============================================
-- 5. Таблица избранного (favorites)
-- ============================================
CREATE TABLE IF NOT EXISTS favorites (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    listing_id INT REFERENCES listings(id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT NOW(),
    UNIQUE(user_id, listing_id)
);
